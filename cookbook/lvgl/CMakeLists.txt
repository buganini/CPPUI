cmake_minimum_required(VERSION 3.24)
project(CPPUICookbookLVGL LANGUAGES C CXX)

set(CPPUI_LVGL_SOURCE_DIR "" CACHE PATH "LVGL source checkout (target v9.6.0; validation pending)")
option(CPPUI_LVGL_SDL "Enable the interactive SDL2 window" OFF)
if(NOT EXISTS "${CPPUI_LVGL_SOURCE_DIR}/lvgl.h")
    message(FATAL_ERROR "Set CPPUI_LVGL_SOURCE_DIR to an LVGL 9 source checkout")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CPPUI_LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
target_compile_definitions(lvgl PUBLIC CPPUI_LVGL_SDL=$<BOOL:${CPPUI_LVGL_SDL}>)
if(CPPUI_LVGL_SDL)
    find_package(SDL2 REQUIRED CONFIG)
    target_link_libraries(lvgl PUBLIC SDL2::SDL2)
endif()
if(UNIX)
    target_link_libraries(lvgl PUBLIC m)
endif()

include(../common.cmake)
cppui_add_cookbook(cppui_cookbook_lvgl lvgl)
target_link_libraries(cppui_cookbook_lvgl PRIVATE lvgl::lvgl)
set_target_properties(cppui_cookbook_lvgl PROPERTIES
    CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)

if(BUILD_TESTING)
    add_executable(cppui_lvgl_lifetime "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/lvgl_lifetime.cpp")
    target_include_directories(cppui_lvgl_lifetime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
    target_link_libraries(cppui_lvgl_lifetime PRIVATE lvgl::lvgl)
    set_target_properties(cppui_lvgl_lifetime PROPERTIES
        CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    cppui_add_test(lvgl.lifetime cppui_lvgl_lifetime)

    if(CPPUI_LVGL_SDL)
        cppui_add_test(cookbook.lvgl_sdl cppui_cookbook_lvgl --smoke --sdl)
        set_property(TEST cookbook.lvgl_sdl APPEND PROPERTY ENVIRONMENT
            "SDL_VIDEODRIVER=dummy" "SDL_AUDIODRIVER=dummy" "SDL_RENDER_DRIVER=software")
        if(CPPUI_VALGRIND_TESTS)
            set_property(TEST valgrind.cookbook.lvgl_sdl APPEND PROPERTY ENVIRONMENT
                "SDL_VIDEODRIVER=dummy" "SDL_AUDIODRIVER=dummy" "SDL_RENDER_DRIVER=software")
        endif()
    endif()

    foreach(header application base button label layout window)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/header_lvgl_${header}.cpp")
        file(WRITE "${source}"
            "#include \"cppui/lvgl/${header}.hpp\"\n#include \"cppui/lvgl/${header}.hpp\"\n")
        add_library(cppui_header_lvgl_${header} OBJECT "${source}")
        target_include_directories(cppui_header_lvgl_${header} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
        target_link_libraries(cppui_header_lvgl_${header} PRIVATE lvgl::lvgl)
    endforeach()
endif()
