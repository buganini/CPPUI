cmake_minimum_required(VERSION 3.24)
project(CPPUIMinimalWx LANGUAGES CXX)
find_package(wxWidgets 3.2 QUIET COMPONENTS core base html OPTIONAL_COMPONENTS webview)
    if(NOT wxWidgets_FOUND)
        find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base html)
    endif()
    if("${wxWidgets_LIBRARIES}" MATCHES "webview")
        add_compile_definitions(CPPUI_WX_WEBVIEW=1)
    endif()
include(${wxWidgets_USE_FILE})
include(../common.cmake)
cppui_add_example(minimal minimal wx "${wxWidgets_LIBRARIES}" example.wx_minimal)
