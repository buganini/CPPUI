cmake_minimum_required(VERSION 3.24)
project(CPPUIVirtualNodes LANGUAGES C CXX)
set(CPPUI_EXAMPLE_BACKEND qt CACHE STRING "Example backend")
set_property(CACHE CPPUI_EXAMPLE_BACKEND PROPERTY STRINGS qt wx lvgl_sdl2 lvgl_x11)
set(backend "${CPPUI_EXAMPLE_BACKEND}")
if(backend STREQUAL "qt")
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    set(toolkit Qt6::Widgets)
elseif(backend STREQUAL "wx")
    find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base html)
    include(${wxWidgets_USE_FILE})
    set(toolkit ${wxWidgets_LIBRARIES})
elseif(backend MATCHES "^lvgl_(sdl2|x11)$")
    include(../../cmake/CPPUILVGLDesktop.cmake)
    cppui_add_lvgl_desktop(${backend})
    set(backend lvgl)
    set(toolkit CPPUI::lvgl)
else()
    message(FATAL_ERROR "Unknown example backend: ${backend}")
endif()
include(../common.cmake)
cppui_add_example(test_virtual_nodes virtual_nodes ${backend} "${toolkit}"
    example.${CPPUI_EXAMPLE_BACKEND}_virtual_nodes ${CPPUI_EXAMPLE_BACKEND})
