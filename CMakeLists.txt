cmake_minimum_required(VERSION 3.24)
project(CPPUI LANGUAGES CXX)

add_library(cppui INTERFACE)
add_library(CPPUI::cppui ALIAS cppui)
target_include_directories(cppui INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(cppui INTERFACE cxx_std_11)

option(CPPUI_BUILD_QT "Build Qt examples and tests (requires Qt 6 Widgets)" OFF)
if(CPPUI_BUILD_QT)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_library(cppui_qt INTERFACE)
    add_library(CPPUI::qt ALIAS cppui_qt)
    target_link_libraries(cppui_qt INTERFACE CPPUI::cppui Qt6::Widgets)

    add_executable(cppui_minimal_qt example/minimal_qt/main.cpp)
    target_link_libraries(cppui_minimal_qt PRIVATE CPPUI::qt)
    add_executable(cppui_virtual_nodes example/test_virtual_nodes/main.cpp)
    target_link_libraries(cppui_virtual_nodes PRIVATE CPPUI::qt)
endif()

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
