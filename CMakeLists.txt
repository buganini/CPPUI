cmake_minimum_required(VERSION 3.24)
project(CPPUI LANGUAGES CXX)

add_library(cppui INTERFACE)
add_library(CPPUI::cppui ALIAS cppui)
target_include_directories(cppui INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(cppui INTERFACE cxx_std_11)

include(CTest)
include(cmake/CPPUItesting.cmake)

option(CPPUI_BUILD_WX "Build wxWidgets examples and tests" OFF)
if(CPPUI_BUILD_WX)
    find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base html)
    include(${wxWidgets_USE_FILE})
    add_library(cppui_wx INTERFACE)
    add_library(CPPUI::wx ALIAS cppui_wx)
    target_compile_definitions(cppui_wx INTERFACE CPPUI_BACKEND_WX=1)
    target_link_libraries(cppui_wx INTERFACE CPPUI::cppui ${wxWidgets_LIBRARIES})
    target_include_directories(cppui_wx INTERFACE ${wxWidgets_INCLUDE_DIRS})
    include(cookbook/common.cmake)
    cppui_add_cookbook(cppui_cookbook_wx wx)
    target_link_libraries(cppui_cookbook_wx_sources PUBLIC CPPUI::wx)
    add_executable(cppui_minimal_wx example/minimal_wx/main.cpp)
    target_link_libraries(cppui_minimal_wx PRIVATE CPPUI::wx)
    if(BUILD_TESTING)
        cppui_add_test(example.wx_minimal cppui_minimal_wx --smoke)
    endif()
endif()

option(CPPUI_BUILD_QT "Build Qt examples and tests (requires Qt 6 Widgets)" OFF)
if(CPPUI_BUILD_QT)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_library(cppui_qt INTERFACE)
    add_library(CPPUI::qt ALIAS cppui_qt)
    target_compile_definitions(cppui_qt INTERFACE CPPUI_BACKEND_QT=1)
    target_link_libraries(cppui_qt INTERFACE CPPUI::cppui Qt6::Widgets)

    include(cookbook/common.cmake)
    cppui_add_cookbook(cppui_cookbook_qt qt)
    target_link_libraries(cppui_cookbook_qt_sources PUBLIC CPPUI::qt)

    add_executable(cppui_minimal_qt example/minimal_qt/main.cpp)
    target_link_libraries(cppui_minimal_qt PRIVATE CPPUI::qt)
    add_executable(cppui_virtual_nodes example/test_virtual_nodes/main.cpp)
    target_link_libraries(cppui_virtual_nodes PRIVATE CPPUI::qt)
    if(BUILD_TESTING)
        cppui_add_test(example.qt_minimal cppui_minimal_qt --smoke)
        cppui_add_test(example.qt_virtual_nodes cppui_virtual_nodes --smoke)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    include(cmake/CPPUItesting.cmake)
    add_subdirectory(tests)
endif()
